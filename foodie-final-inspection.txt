import { NextRequest, NextResponse, connection } from "next/server";
import { prisma } from "@/lib/prisma";
import { getCurrentUser } from "@/lib/auth";

// Get cart items
export async function GET(request: NextRequest) {
  await connection();

  try {
    const user = await getCurrentUser(request);

    if (!user) {
      return NextResponse.json(
        { message: "Please login first." },
        { status: 401 }
      );
    }

    const items = await prisma.cartItem.findMany({
      where: {
        cart: { userId: user.id },
      },
      select: {
        id: true,
        idMeal: true,
        name: true,
        price: true,
        image: true,
        category: true,
        quantity: true,
      },
    });

    const formattedItems = items.map((item) => ({
      ...item,
      price: item.price.toString(),
      image: item.image || "",
    }));

    return NextResponse.json(
      { items: formattedItems },
      {
        headers: {
          "Cache-Control": "no-store",
        },
      }
    );
  } catch (error) {
    console.error("Cart fetch error:", error);

    return NextResponse.json(
      { message: "Unable to fetch cart." },
      { status: 500 }
    );
  }
}

// Add item to cart
export async function POST(request: NextRequest) {
  await connection();

  try {
    const user = await getCurrentUser(request);

    if (!user) {
      return NextResponse.json(
        { message: "Please login first." },
        { status: 401 }
      );
    }

    const body = await request.json();

    const name = String(body.name || "").trim();
    const price = Number(
      String(body.price || "").replace(/[^\d.]/g, "")
    );
    const image = String(body.image || "");
    const idMeal = body.idMeal ? String(body.idMeal) : null;
    const category = body.category ? String(body.category) : null;

    if (!name || !Number.isFinite(price) || price <= 0) {
      return NextResponse.json(
        { message: "Invalid food details." },
        { status: 400 }
      );
    }

    const cart = await prisma.cart.upsert({
      where: { userId: user.id },
      create: { userId: user.id },
      update: {},
    });

    const existingItem = await prisma.cartItem.findFirst({
      where: {
        cartId: cart.id,
        ...(idMeal ? { idMeal } : { name }),
      },
    });

    const savedItem = existingItem
      ? await prisma.cartItem.update({
          where: { id: existingItem.id },
          data: {
            quantity: { increment: 1 },
            price,
            image,
            category,
          },
        })
      : await prisma.cartItem.create({
          data: {
            cartId: cart.id,
            idMeal,
            name,
            price,
            image,
            category,
            quantity: 1,
          },
        });

    return NextResponse.json(
      {
        message: "Item added to cart.",
        item: {
          id: savedItem.id,
          idMeal: savedItem.idMeal,
          name: savedItem.name,
          price: savedItem.price.toString(),
          image: savedItem.image || "",
          category: savedItem.category,
          quantity: savedItem.quantity,
        },
      },
      { status: 201 }
    );
  } catch (error) {
    console.error("Cart add error:", error);

    return NextResponse.json(
      { message: "Unable to add item to cart." },
      { status: 500 }
    );
  }
}

      where: { id },

import { NextRequest, NextResponse, connection } from "next/server";
import { createHash } from "crypto";
import { prisma } from "@/lib/prisma";

export async function GET(request: NextRequest) {
  await connection();

  try {
    const token = request.cookies.get("fooddie_session")?.value;

    if (!token) {
      return NextResponse.json(
        { user: null },
        { status: 401 }
      );
    }

    const tokenHash = createHash("sha256")
      .update(token)
      .digest("hex");

    const session = await prisma.session.findUnique({
      where: { tokenHash },
      include: { user: true },
    });

    if (!session || session.expiresAt <= new Date()) {
      if (session) {
        await prisma.session.delete({
          where: { id: session.id },
        });
      }

      const response = NextResponse.json(
        { user: null },
        { status: 401 }
      );

      response.cookies.set("fooddie_session", "", {
        httpOnly: true,
        secure: process.env.NODE_ENV === "production",
        sameSite: "lax",
        expires: new Date(0),
        path: "/",
      });

      return response;
    }

    return NextResponse.json(
      {
        user: {
          name: session.user.name,
          email: session.user.email,
          role: session.user.role,
        },
      },
      {
        headers: {
          "Cache-Control": "no-store",
        },
      }
    );
  } catch (error) {
    console.error("Session verification error:", error);

    return NextResponse.json(
      { message: "Unable to verify session" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse, connection } from "next/server";
import { getAdminWithPermission } from "@/lib/admin-permission";
import { PERMISSIONS } from "@/lib/permissions";
import { prisma } from "@/lib/prisma";

export async function GET(request: NextRequest) {
  await connection();

  try {
    const admin = await getAdminWithPermission(
      request,
      PERMISSIONS.DASHBOARD
    );

    if (!admin) {
      return NextResponse.json(
        { message: "Admin access denied" },
        { status: 403 }
      );
    }

    const [users, orders, pendingOrders, revenue] =
      await Promise.all([
        prisma.user.count(),
        prisma.order.count(),
        prisma.order.count({
          where: { status: "Pending" },
        }),
        prisma.order.aggregate({
          _sum: { total: true },
        }),
      ]);

    return NextResponse.json({
      success: true,
      admin: {
        name: admin.name,
        email: admin.email,
      },
      statistics: {
        users,
        orders,
        pendingOrders,
        revenue: Number(revenue._sum.total || 0),
      },
    });
  } catch (error) {
    console.error("Dashboard error:", error);

    return NextResponse.json(
      { message: "Unable to load dashboard" },
      { status: 500 }
    );
  }
}

"use client";

import { useEffect, useState } from "react";

export type CartItem = {
  id: string;
  idMeal?: string | null;
  name: string;
  price: string;
  image: string;
  category?: string | null;
  quantity: number;
};

export function useCart() {
  const [cart, setCart] = useState<CartItem[]>([]);

  // Fetch cart from database
  const fetchCart = async () => {
    try {
      const response = await fetch("/api/cart", {
        cache: "no-store",
      });

      if (!response.ok) {
        setCart([]);
        return;
      }

      const data = await response.json();
      setCart(data.items);
    } catch (error) {
      console.error("Cart fetch error:", error);
    }
  };

  useEffect(() => {
    fetchCart();

    // Listen for cart changes
    window.addEventListener("fooddie-cart-change", fetchCart);

    return () => {
      window.removeEventListener("fooddie-cart-change", fetchCart);
    };
  }, []);

  // Add item
  const addItem = async (item: Omit<CartItem, "id" | "quantity">) => {
    try {
      const response = await fetch("/api/cart", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify(item),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.message || "Unable to add item.");
      }

      window.dispatchEvent(new Event("fooddie-cart-change"));

      return true;
    } catch (error) {
      console.error("Add cart error:", error);
      return false;
    }
  };

  // Update quantity
  const updateQuantity = async (id: string, quantity: number) => {
    if (quantity <= 0) {
      await removeItem(id);
      return;
    }

    const response = await fetch(`/api/cart/${id}`, {
      method: "PATCH",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ quantity }),
    });

    if (response.ok) {
      window.dispatchEvent(new Event("fooddie-cart-change"));
    }
  };

  // Remove item
  const removeItem = async (id: string) => {
    const response = await fetch(`/api/cart/${id}`, {
      method: "DELETE",
    });

    if (response.ok) {
      window.dispatchEvent(new Event("fooddie-cart-change"));
    }
  };

  return {
    cart,
    addItem,
    updateQuantity,
    removeItem,
    clearCart: fetchCart,
  };
}
